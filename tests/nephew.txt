pokew boot_campaign campaign_single

run 1 fire                      ; title: start

until game_state 0 -

goto 1 0 21 right               ; into the east wing, ground floor

expect selfie_index 1

goto 1 0 21 right
expect npc_done 0
until pl_state 4 right 300
